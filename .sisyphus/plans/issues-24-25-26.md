```rust
// Handle lines() function: lines($VAR) > N or lines($...VAR) > N
// Must be checked BEFORE generic comparison to avoid string-comparison fallback
if expr.contains("lines(") {
    // Parse patterns like: lines($VAR) > N, lines($...STMTS) > N, lines($VAR) < N, etc.
    for op in &["!=", "==", ">=", "<=", ">", "<"] {
        if let Some(pos) = expr.find(op) {
            let left = expr[..pos].trim();
            let right = expr[pos + op.len()..].trim();
            // Extract the metavariable name from lines($VAR) or lines($...VAR)
            if let Some(inner) = left.strip_prefix("lines(").and_then(|s| s.strip_suffix(")")) {
                let metavar = inner.trim();
                // Resolve the bound value: if metavar is $VAR or $...VAR,
                // the `value` parameter already holds the bound text
                let bound_text = if metavar.starts_with('$') {
                    value  // evaluate_python_expression receives the bound value
                } else {
                    value
                };
                let line_count = bound_text.lines().count();
                if let Ok(threshold) = right.parse::<usize>() {
                    let result = match *op {
                        "==" => line_count == threshold,
                        "!=" => line_count != threshold,
                        ">" => line_count > threshold,
                        "<" => line_count < threshold,
                        ">=" => line_count >= threshold,
                        "<=" => line_count <= threshold,
                        _ => false,
                    };
                    tracing::debug!(
                        "DEBUG lines(): text has {} lines, threshold {}, op '{}', result={}",
                        line_count, threshold, op, result
                    );
                    return Ok(result);
                }
            }
        }
    }
}
```

```rust
//! Plain text language parser for astgrep
//!
//! Does NOT use tree-sitter. Wraps text content in a minimal UniversalNode
//! so that `pattern-regex` and string-based patterns can match against raw text.

use astgrep_ast::UniversalNode;
use astgrep_ast::NodeType;
use astgrep_core::{AstNode, Language, LanguageParser, Result};
use std::path::Path;

/// Plain text parser — creates a single-node AST wrapping the file content.
///
/// The root node is a `Program` with the full text as its `text` field.
/// Each line is also added as a child `ExpressionStatement` for line-level matching.
pub struct TextParser;

impl TextParser {
    pub fn new() -> Self {
        Self
    }
}

impl Default for TextParser {
    fn default() -> Self {
        Self::new()
    }
}

impl LanguageParser for TextParser {
    fn parse(&self, source: &str, file_path: &Path) -> Result<Box<dyn AstNode>> {
        let mut root = UniversalNode::new(NodeType::Program)
            .with_text(source.to_string())
            .with_attribute("file_path".to_string(), file_path.to_string_lossy().to_string());

        // Add each line as a child for line-level pattern matching
        for (line_idx, line) in source.lines().enumerate() {
            let line_node = UniversalNode::new(NodeType::ExpressionStatement)
                .with_text(line.to_string())
                .with_location(line_idx + 1, 1, line_idx + 1, line.len() + 1);
            root.children.push(line_node);
        }

        Ok(Box::new(root))
    }

    fn language(&self) -> Language {
        Language::Text
    }

    fn supports_file(&self, file_path: &Path) -> bool {
        Language::Text
            .extensions()
            .iter()
            .any(|ext| file_path.to_string_lossy().ends_with(ext))
    }
}
```

```rust
            if s.contains('$') || s.contains("...") {
                return Err("Text language supports only pattern-regex, \
                           not structural patterns with metavariables/ellipsis".into());
            }
        }
    }
}
```
